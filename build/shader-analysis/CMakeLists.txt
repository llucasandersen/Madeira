cmake_minimum_required(VERSION 3.20)
project(MadeiraShaderAnalysis LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(LLVM_TARGETS_TO_BUILD "" CACHE STRING "" FORCE)
set(LLVM_ENABLE_PROJECTS "" CACHE STRING "" FORCE)
set(LLVM_INCLUDE_TESTS OFF CACHE BOOL "" FORCE)
set(LLVM_INCLUDE_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLVM_INCLUDE_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(LLVM_ENABLE_ZLIB OFF CACHE BOOL "" FORCE)
set(LLVM_ENABLE_ZSTD OFF CACHE BOOL "" FORCE)
set(LLVM_ENABLE_TERMINFO OFF CACHE BOOL "" FORCE)
set(LLVM_USE_CRT_RELEASE MT CACHE STRING "" FORCE)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
add_subdirectory("${LLVM_SOURCE}/llvm" llvm)
add_executable(shader-volume-rewrite ../../tools/shader-volume-rewrite.cpp ../../madeira-d3d12/src/unix/madeira_uint_volumes.cpp)
target_include_directories(shader-volume-rewrite PRIVATE "${LLVM_SOURCE}/llvm/include" "${CMAKE_CURRENT_BINARY_DIR}/llvm/include")
target_compile_definitions(shader-volume-rewrite PRIVATE _CRT_SECURE_NO_WARNINGS)
target_link_libraries(shader-volume-rewrite PRIVATE LLVMBitReader LLVMBitWriter LLVMCore LLVMSupport)
